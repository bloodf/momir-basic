function mapNativeTypeToTransport(type: string): PrinterTransport {
  const upperType = type.toUpperCase();
  if (upperType.includes('BLE') || upperType.includes('BLUETOOTH_LE')) return 'ble';
  if (upperType.includes('CLASSIC') || upperType.includes('SPP')) return 'classic';

## TCP Validation (lines 42-69)

### validateTcpAddress() - Explicit host:port validation
```typescript
function validateTcpAddress(address: string): void {
  if (!address.includes(':')) {
    throw new PrinterAdapterError(PrinterErrorCode.TCP_INVALID_ADDRESS, ...);
  }
  const colonIndex = address.lastIndexOf(':');
  const host = address.substring(0, colonIndex);
  const portStr = address.substring(colonIndex + 1);
  const port = parseInt(portStr, 10);

  if (!host || host.trim().length === 0) {
    throw new PrinterAdapterError(PrinterErrorCode.TCP_INVALID_ADDRESS, ...);
  }
  if (isNaN(port) || port <= 0 || port > 65535) {
    throw new PrinterAdapterError(PrinterErrorCode.TCP_INVALID_ADDRESS, ...);
  }
}
```
- Validates host is non-empty
- Validates port is 1-65535
- Separate from Bluetooth validation
